local function fire(id, kind)
	id = tonumber(id)
	if not id then return false end

--============================ http ============================

local function httpGet(url)
	local ok, res = pcall(function()
		if typeof(request) == "function" then return request({Url = url, Method = "GET"}).Body end
		if typeof(http_request) == "function" then return http_request({Url = url, Method = "GET"}).Body end
		if typeof(syn) == "table" and syn.request then return syn.request({Url = url}).Body end
		if typeof(http) == "table" and http.request then return http.request({Url = url}).Body end
		return game:HttpGet(url)
	end)
	if ok and type(res) == "string" and #res > 2 then return res end
	return nil
end

local function getJson(url)
	local body = httpGet(url)
	if body then
		local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
		if ok and type(data) == "table" and not data.errors then return data end
	end
	local alt = url:gsub("^https://games%.roblox%.com", "https://games.roproxy.com")
		:gsub("^https://apis%.roblox%.com", "https://apis.roproxy.com")
	if alt ~= url then
		local b2 = httpGet(alt)
		if b2 then
			local ok, data = pcall(function() return HttpService:JSONDecode(b2) end)
			if ok and type(data) == "table" and not data.errors then return data end
		end
	end
	return nil
end

local function fetchPasses()
	local out, cursor = {}, nil
	for _ = 1, 10 do
		local url = "https://games.roblox.com/v1/games/" .. tostring(game.GameId) ..
			"/game-passes?limit=100&sortOrder=Asc"
		if cursor and cursor ~= "" then url = url .. "&cursor=" .. cursor end
		local data = getJson(url)
		if not data or type(data.gamePasses) ~= "table" then break end
		for _, p in ipairs(data.gamePasses) do
			if p.id then
				out[#out + 1] = {
					id = p.id, name = p.name or ("Pass " .. tostring(p.id)),
					price = tonumber(p.price) or 0, kind = "Gamepass",
					icon = "rbxthumb://type=GamePass&id=" .. tostring(p.id) .. "&w=150&h=150",
				}
			end
		end
		cursor = data.nextPageCursor
		if not cursor or cursor == "" then break end
	end
	return out
end

local function fetchProducts()
	local out = {}
	local ok, pages = pcall(function() return ms:GetDeveloperProductsAsync() end)
	if ok and pages then
		pcall(function()
			while true do
				for _, p in ipairs(pages:GetCurrentPage()) do
					out[#out + 1] = {id = p.ProductId, name = p.Name or ("Product " .. tostring(p.ProductId)),
						price = tonumber(p.PriceInRobux) or 0, kind = "Product"}
				end
				if pages.IsFinished then break end
				pages:AdvanceToNextPageAsync()
			end
		end)
		if #out > 0 then return out end
	end
	local page = 1
	for _ = 1, 10 do
		local data = getJson("https://apis.roblox.com/developer-products/v2/universes/" ..
			tostring(game.GameId) .. "/developerproducts?limit=100&page=" .. tostring(page))
		if not data or type(data.developerProducts) ~= "table" then break end
		for _, p in ipairs(data.developerProducts) do
			local pid = p.productId or p.id
			if pid then
				out[#out + 1] = {id = pid, name = p.name or ("Product " .. tostring(pid)),
					price = tonumber(p.priceInRobux) or 0, kind = "Product"}
			end
		end
		page = page + 1
		if page > (tonumber(data.totalPages) or 1) then break end
	end
	return out
end

local function drag(frame, handle, onClick)
	handle = handle or frame
	local dragging, moved, startPos, startInput = false, false, nil, nil
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging, moved = true, false
			startInput, startPos = input.Position, frame.Position
			local conn
			conn = input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false; conn:Disconnect()
					if onClick and not moved then onClick() end
				end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - startInput
			if math.abs(d.X) + math.abs(d.Y) > 6 then moved = true end
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
end

end

local function srcOfConn(c)
	local ok, s = pcall(function() return c.Script end)
	if ok and typeof(s) == "Instance" then
		local ok2, p = pcall(function() return s:GetFullName() end)
		return ok2 and p or s.Name
	end
	local f
	pcall(function() f = c.Function end)
	if type(f) == "function" then
		local ok3, src = pcall(function() return debug.info(f, "s") end)
		if ok3 and type(src) == "string" then return src end
	end
	return "?"
end

local function isCore(src) return src:find("CoreScripts") ~= nil end

local function gameListeners(sig)
	if type(getconnections) ~= "function" then return 0, {} end
	local ok, conns = pcall(getconnections, sig)
	if not ok or type(conns) ~= "table" then return 0, {} end
	local n, list = 0, {}
	for _, c in ipairs(conns) do
		local src = srcOfConn(c)
		if not isCore(src) then n = n + 1; list[#list + 1] = c end
	end
	return n, list
end

local VCOL = {
	ok      = {bg = Color3.fromRGB(36, 68, 46),  fg = Color3.fromRGB(140, 230, 165)},
	maybe   = {bg = Color3.fromRGB(72, 62, 34),  fg = Color3.fromRGB(245, 205, 110)},
	bad     = {bg = Color3.fromRGB(74, 34, 40),  fg = Color3.fromRGB(255, 130, 130)},
	unknown = {bg = COL.soft,                    fg = COL.dim},
}

local reactionSeen, hooksInstalled, hooked = false, 0, {}
local lastVerdict = ""

local function setVerdict(state, text)
	if reactionSeen and state == "bad" then state, text = "ok", "пойдет нормас — игра ответила" end
	if text == lastVerdict then return end
	lastVerdict = text
	local c = VCOL[state] or VCOL.unknown
	verdict.Text = text
	verdict.TextColor3 = c.fg
	verdict.BackgroundColor3 = c.bg
	print("[ВЕРДИКТ] " .. text)
end

local function evaluate()
	if type(getconnections) ~= "function" then
		setVerdict("unknown", "getconnections нет — не проверить")
		return
	end
	local g = gameListeners(ms.PromptGamePassPurchaseFinished)
	local p = gameListeners(ms.PromptProductPurchaseFinished)
	print(("[ПРОВЕРКА] слушателей игры: gamepass=%d product=%d"):format(g, p))
	if g > 0 then setVerdict("ok", "пойдет нормас")
	elseif p > 0 then setVerdict("maybe", "пойдет нормас (только продукты)")
	else setVerdict("bad", "нахуй игра не подходит") end
end

local function installHooks()
	if type(hookfunction) ~= "function" or type(getconnections) ~= "function" then return end
	local sigs = {
		Gamepass = ms.PromptGamePassPurchaseFinished,
		Product = ms.PromptProductPurchaseFinished,
		Asset = ms.PromptPurchaseFinished,
	}
	for name, sig in pairs(sigs) do
		local _, conns = gameListeners(sig)
		for _, c in ipairs(conns) do
			if not hooked[c] then
				local f
				pcall(function() f = c.Function end)
				if type(f) == "function" then
					local ok = pcall(function()
						c.Function = hookfunction(f, function(...)
							reactionSeen = true
							setVerdict("ok", "пойдет нормас — игра ответила")
							return f(...)
						end)
					end)
					if ok then hooked[c] = true; hooksInstalled = hooksInstalled + 1 end
				end
			end
		end
	end
end

local function installLoggers()
	if type(hookfunction) ~= "function" then return end
	pcall(function()
		hookfunction(ms.UserOwnsGamePassAsync, function(self, userId, passId)

			return res
		end)
	end)
	pcall(function()
		hookfunction(ms.PromptGamePassPurchase, function(self, player, passId)
			print("[ИГРА ПРОМПТИТ] пасс " .. tostring(passId))
			return ms.PromptGamePassPurchase(self, player, passId)
		end)
	end)
end

		for _, p in ipairs(passes) do all[#all + 1] = p end
		for _, p in ipairs(products) do all[#all + 1] = p end
		table.sort(all, function(a, b) return (a.price or 0) < (b.price or 0) end)

		local n = 0
		for i, d in ipairs(all) do
			if d.id then n = n + 1; addRow(d, i) end
		end

		footLeft.Text = "найдено: " .. n
		if n > 0 then
			status.Text = "клик по строке → ID в поле → FIRE"
			status.TextColor3 = COL.dim
		else
			status.Text = "в игре нет пассов/продуктов"
			status.TextColor3 = COL.accent
		end
		scanning = false
	end)
end

	installHooks()
	scan()
end)

btnFire.Activated:Connect(function()
	local id = tonumber((tostring(input.Text):gsub("%s", "")))
	if not id then
		status.Text = "нужен числовой ID"
		status.TextColor3 = COL.accent
		return
	end
	local kind = kindOf[tostring(id)]
	local ok = fire(id, kind)
	if ok then
		status.Text = "FIRE → " .. tostring(id) .. (kind and (" · " .. kind) or "")
		status.TextColor3 = COL.green
	else
		status.Text = "не смог выстрелить"
		status.TextColor3 = COL.accent
	end
end)
